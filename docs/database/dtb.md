-- =========================================================================
-- LECTURER EVALUATION SYSTEM
-- PostgreSQL 15 / EF Core 8 schema — English version
--
-- All identifiers are PascalCase and CASE-SENSITIVE -> always wrap in double
-- quotes in hand-written SQL: SELECT * FROM "Courses" WHERE "CourseCode" = '...'
--
-- Run this file top to bottom to create an empty database with the correct
-- structure — tables are ordered by foreign key dependency.
--
-- No extensions required (gen_random_uuid() is built into PostgreSQL 13+).
-- =========================================================================


-- ================= 1. ORGANIZATION =================

CREATE TABLE public."Faculties" (
    "FacultyId"   integer GENERATED BY DEFAULT AS IDENTITY,
    "FacultyName" text NOT NULL,
    CONSTRAINT "PK_Faculties" PRIMARY KEY ("FacultyId")
);
CREATE UNIQUE INDEX "IX_Faculties_FacultyName" ON public."Faculties" ("FacultyName");

CREATE TABLE public."Departments" (
    "DepartmentId"   integer GENERATED BY DEFAULT AS IDENTITY,
    "DepartmentName" text NOT NULL,
    "FacultyId"      integer NULL,
    CONSTRAINT "PK_Departments" PRIMARY KEY ("DepartmentId"),
    CONSTRAINT "FK_Departments_Faculties_FacultyId" FOREIGN KEY ("FacultyId")
        REFERENCES public."Faculties" ("FacultyId") ON DELETE SET NULL
);
CREATE INDEX "IX_Departments_FacultyId" ON public."Departments" ("FacultyId");


-- ================= 2. LOGIN (Admin accounts only) =================

CREATE TABLE public."Roles" (
    "RoleId"   integer GENERATED BY DEFAULT AS IDENTITY,
    "RoleName" text NOT NULL,
    CONSTRAINT "PK_Roles" PRIMARY KEY ("RoleId")
);
CREATE UNIQUE INDEX "IX_Roles_RoleName" ON public."Roles" ("RoleName");

CREATE TABLE public."Accounts" (
    "AccountId"     integer GENERATED BY DEFAULT AS IDENTITY,
    "Username"      text NOT NULL,
    "PasswordHash"  text NOT NULL,
    "RoleId"        integer NOT NULL,
    "IsActive"      boolean NOT NULL,
    "CreatedAt"     timestamp with time zone NOT NULL,
    CONSTRAINT "PK_Accounts" PRIMARY KEY ("AccountId"),
    CONSTRAINT "FK_Accounts_Roles_RoleId" FOREIGN KEY ("RoleId")
        REFERENCES public."Roles" ("RoleId") ON DELETE CASCADE
);
CREATE UNIQUE INDEX "IX_Accounts_Username" ON public."Accounts" ("Username");
CREATE INDEX        "IX_Accounts_RoleId"   ON public."Accounts" ("RoleId");


-- ================= 3. LECTURERS (log in via Gmail, not linked to Accounts) =================

CREATE TABLE public."Lecturers" (
    "LecturerId"  integer GENERATED BY DEFAULT AS IDENTITY,
    "FullName"    text NOT NULL,
    "DepartmentId" integer NULL,
    "FacultyId"   integer NULL,
    "Email"       text NULL,
    "PhoneNumber" text NULL,
    CONSTRAINT "PK_Lecturers" PRIMARY KEY ("LecturerId"),
    CONSTRAINT "FK_Lecturers_Departments_DepartmentId" FOREIGN KEY ("DepartmentId")
        REFERENCES public."Departments" ("DepartmentId") ON DELETE RESTRICT,
    CONSTRAINT "FK_Lecturers_Faculties_FacultyId" FOREIGN KEY ("FacultyId")
        REFERENCES public."Faculties" ("FacultyId") ON DELETE RESTRICT
);
CREATE UNIQUE INDEX "IX_Lecturers_Email"        ON public."Lecturers" ("Email");
CREATE INDEX        "IX_Lecturers_DepartmentId" ON public."Lecturers" ("DepartmentId");
CREATE INDEX        "IX_Lecturers_FacultyId"    ON public."Lecturers" ("FacultyId");


-- ================= 4. CURRICULUM =================

CREATE TABLE public."Majors" (
    "MajorId"   integer GENERATED BY DEFAULT AS IDENTITY,
    "MajorName" text NOT NULL,
    "FacultyId" integer NOT NULL,
    CONSTRAINT "PK_Majors" PRIMARY KEY ("MajorId"),
    CONSTRAINT "FK_Majors_Faculties_FacultyId" FOREIGN KEY ("FacultyId")
        REFERENCES public."Faculties" ("FacultyId") ON DELETE CASCADE
);
CREATE INDEX "IX_Majors_FacultyId" ON public."Majors" ("FacultyId");

CREATE TABLE public."Curricula" (
    "CurriculumId"       integer GENERATED BY DEFAULT AS IDENTITY,
    "MajorId"            integer NOT NULL,
    "TotalCredits"       integer NOT NULL,
    "RequiredCredits"    integer NOT NULL,
    "MinElectiveCredits" integer NOT NULL,
    CONSTRAINT "PK_Curricula" PRIMARY KEY ("CurriculumId"),
    CONSTRAINT "FK_Curricula_Majors_MajorId" FOREIGN KEY ("MajorId")
        REFERENCES public."Majors" ("MajorId") ON DELETE CASCADE
);
-- UNIQUE => each major has exactly one curriculum
CREATE UNIQUE INDEX "IX_Curricula_MajorId" ON public."Curricula" ("MajorId");


-- ================= 5. COURSES =================

CREATE TABLE public."Courses" (
    "CourseId"             integer GENERATED BY DEFAULT AS IDENTITY,
    "CourseCode"           text NOT NULL,           -- manually entered, e.g. '19783', '19783H'
    "CourseName"           text NOT NULL,
    "Credits"              integer NOT NULL,
    "CourseType"           text NOT NULL DEFAULT '',   -- Required | Elective
    "DepartmentId"         integer NULL,
    "FacultyId"            integer NULL,            -- course belongs directly to a faculty
    "PrerequisiteCourseId" integer NULL,             -- self-reference, at most one prerequisite
    CONSTRAINT "PK_Courses" PRIMARY KEY ("CourseId"),
    CONSTRAINT "FK_Courses_Departments_DepartmentId" FOREIGN KEY ("DepartmentId")
        REFERENCES public."Departments" ("DepartmentId") ON DELETE RESTRICT,
    CONSTRAINT "FK_Courses_Faculties_FacultyId" FOREIGN KEY ("FacultyId")
        REFERENCES public."Faculties" ("FacultyId") ON DELETE RESTRICT,
    CONSTRAINT "FK_Courses_Courses_PrerequisiteCourseId" FOREIGN KEY ("PrerequisiteCourseId")
        REFERENCES public."Courses" ("CourseId") ON DELETE RESTRICT
);
CREATE UNIQUE INDEX "IX_Courses_CourseCode"           ON public."Courses" ("CourseCode");
CREATE INDEX        "IX_Courses_DepartmentId"         ON public."Courses" ("DepartmentId");
CREATE INDEX        "IX_Courses_FacultyId"            ON public."Courses" ("FacultyId");
CREATE INDEX        "IX_Courses_PrerequisiteCourseId" ON public."Courses" ("PrerequisiteCourseId");

CREATE TABLE public."CurriculumCourses" (
    "Id"           integer GENERATED BY DEFAULT AS IDENTITY,
    "CurriculumId" integer NOT NULL,
    "CourseId"     integer NOT NULL,
    "SemesterOrder" integer NOT NULL,       -- intended semester within the curriculum (1..n)
    CONSTRAINT "PK_CurriculumCourses" PRIMARY KEY ("Id"),
    CONSTRAINT "FK_CurriculumCourses_Curricula_CurriculumId" FOREIGN KEY ("CurriculumId")
        REFERENCES public."Curricula" ("CurriculumId") ON DELETE CASCADE,
    CONSTRAINT "FK_CurriculumCourses_Courses_CourseId" FOREIGN KEY ("CourseId")
        REFERENCES public."Courses" ("CourseId") ON DELETE CASCADE
);
CREATE UNIQUE INDEX "IX_CurriculumCourses_CurriculumId_CourseId"
    ON public."CurriculumCourses" ("CurriculumId", "CourseId");
CREATE INDEX "IX_CurriculumCourses_CourseId" ON public."CurriculumCourses" ("CourseId");


-- ================= 6. ACADEMIC CALENDAR & COURSE SECTIONS =================

CREATE TABLE public."AcademicYears" (
    "AcademicYearId"   integer GENERATED BY DEFAULT AS IDENTITY,
    "AcademicYearName" text NOT NULL,
    "StartDate"        date NOT NULL,
    "EndDate"          date NOT NULL,
    CONSTRAINT "PK_AcademicYears" PRIMARY KEY ("AcademicYearId")
);
CREATE UNIQUE INDEX "IX_AcademicYears_AcademicYearName" ON public."AcademicYears" ("AcademicYearName");

CREATE TABLE public."Semesters" (
    "SemesterId"     integer GENERATED BY DEFAULT AS IDENTITY,
    "SemesterName"   text NOT NULL,
    "AcademicYearId" integer NOT NULL,
    CONSTRAINT "PK_Semesters" PRIMARY KEY ("SemesterId"),
    CONSTRAINT "FK_Semesters_AcademicYears_AcademicYearId" FOREIGN KEY ("AcademicYearId")
        REFERENCES public."AcademicYears" ("AcademicYearId") ON DELETE CASCADE
);
CREATE UNIQUE INDEX "IX_Semesters_AcademicYearId_SemesterName"
    ON public."Semesters" ("AcademicYearId", "SemesterName");

CREATE TABLE public."CourseSections" (
    "CourseSectionId" integer GENERATED BY DEFAULT AS IDENTITY,
    "CourseId"        integer NOT NULL,
    "SemesterId"      integer NOT NULL,
    "LecturerId"      integer NOT NULL,   -- mỗi lớp học phần có đúng một giảng viên
    "SectionName"     text NOT NULL,
    "ClassSize"       integer NOT NULL,
    CONSTRAINT "PK_CourseSections" PRIMARY KEY ("CourseSectionId"),
    CONSTRAINT "FK_CourseSections_Courses_CourseId" FOREIGN KEY ("CourseId")
        REFERENCES public."Courses" ("CourseId") ON DELETE CASCADE,
    CONSTRAINT "FK_CourseSections_Semesters_SemesterId" FOREIGN KEY ("SemesterId")
        REFERENCES public."Semesters" ("SemesterId") ON DELETE CASCADE,
    CONSTRAINT "FK_CourseSections_Lecturers_LecturerId" FOREIGN KEY ("LecturerId")
        REFERENCES public."Lecturers" ("LecturerId") ON DELETE RESTRICT
);
CREATE UNIQUE INDEX "IX_CourseSections_CourseId_SemesterId_SectionName"
    ON public."CourseSections" ("CourseId", "SemesterId", "SectionName");
CREATE INDEX "IX_CourseSections_SemesterId" ON public."CourseSections" ("SemesterId");
CREATE INDEX "IX_CourseSections_LecturerId" ON public."CourseSections" ("LecturerId");


-- ================= 7. ANSWER SCALES (reusable across survey templates) =================

CREATE TABLE public."AnswerScales" (
    "AnswerScaleId"   integer GENERATED BY DEFAULT AS IDENTITY,
    "AnswerScaleName" text NOT NULL,        -- e.g. 'Satisfaction scale', 'Percentage scale'
    CONSTRAINT "PK_AnswerScales" PRIMARY KEY ("AnswerScaleId")
);

CREATE TABLE public."AnswerScaleOptions" (
    "AnswerScaleOptionId" integer GENERATED BY DEFAULT AS IDENTITY,
    "AnswerScaleId"       integer NOT NULL,
    "Value"               integer NOT NULL,    -- 1..5
    "DisplayText"         text NOT NULL,        -- e.g. 'Dissatisfied' or '20%'
    CONSTRAINT "PK_AnswerScaleOptions" PRIMARY KEY ("AnswerScaleOptionId"),
    CONSTRAINT "FK_AnswerScaleOptions_AnswerScales_AnswerScaleId" FOREIGN KEY ("AnswerScaleId")
        REFERENCES public."AnswerScales" ("AnswerScaleId") ON DELETE CASCADE,
    CONSTRAINT "CK_AnswerScaleOptions_Value" CHECK ("Value" BETWEEN 1 AND 5)
);
-- each value 1..5 appears at most once per answer scale
CREATE UNIQUE INDEX "IX_AnswerScaleOptions_AnswerScaleId_Value"
    ON public."AnswerScaleOptions" ("AnswerScaleId", "Value");


-- ================= 8. SURVEY TEMPLATES (Admin-authored, max 30 questions each) =================

CREATE TABLE public."SurveyTemplates" (
    "SurveyTemplateId" integer GENERATED BY DEFAULT AS IDENTITY,
    "TemplateName"      text NOT NULL,
    "AnswerScaleId"     integer NOT NULL,     -- the whole template shares one answer scale
    "CreatedAt"         timestamp with time zone NOT NULL,
    CONSTRAINT "PK_SurveyTemplates" PRIMARY KEY ("SurveyTemplateId"),
    CONSTRAINT "FK_SurveyTemplates_AnswerScales_AnswerScaleId" FOREIGN KEY ("AnswerScaleId")
        REFERENCES public."AnswerScales" ("AnswerScaleId") ON DELETE RESTRICT
);
CREATE INDEX "IX_SurveyTemplates_AnswerScaleId" ON public."SurveyTemplates" ("AnswerScaleId");

CREATE TABLE public."SurveyQuestions" (
    "QuestionId"       integer GENERATED BY DEFAULT AS IDENTITY,
    "SurveyTemplateId" integer NOT NULL,
    "QuestionText"     text NOT NULL,
    CONSTRAINT "PK_SurveyQuestions" PRIMARY KEY ("QuestionId"),
    CONSTRAINT "FK_SurveyQuestions_SurveyTemplates_SurveyTemplateId" FOREIGN KEY ("SurveyTemplateId")
        REFERENCES public."SurveyTemplates" ("SurveyTemplateId") ON DELETE CASCADE
);
CREATE INDEX "IX_SurveyQuestions_SurveyTemplateId" ON public."SurveyQuestions" ("SurveyTemplateId");

-- "Max 30 questions per template" cannot be expressed with a plain CHECK/FK
-- (CHECK cannot see other rows) -> enforced with a trigger.
-- Remove this trigger and validate in the backend instead if you prefer a simpler DB.
CREATE OR REPLACE FUNCTION public.fn_limit_survey_questions()
RETURNS trigger AS $$
BEGIN
    IF (SELECT COUNT(*) FROM public."SurveyQuestions"
        WHERE "SurveyTemplateId" = NEW."SurveyTemplateId") >= 30 THEN
        RAISE EXCEPTION 'A survey template can have at most 30 questions';
    END IF;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_limit_survey_questions
    BEFORE INSERT ON public."SurveyQuestions"
    FOR EACH ROW EXECUTE FUNCTION public.fn_limit_survey_questions();


-- ================= 9. SEMESTER + PER-SECTION SURVEYS (unique link/QR per section) =================

CREATE TABLE public."SemesterSurveys" (
    "SemesterSurveyId" integer GENERATED BY DEFAULT AS IDENTITY,
    "SemesterId"       integer NOT NULL,
    "SurveyTemplateId" integer NOT NULL,
    "CreatedAt"        timestamp with time zone NOT NULL,
    CONSTRAINT "PK_SemesterSurveys" PRIMARY KEY ("SemesterSurveyId"),
    CONSTRAINT "FK_SemesterSurveys_Semesters_SemesterId" FOREIGN KEY ("SemesterId")
        REFERENCES public."Semesters" ("SemesterId") ON DELETE CASCADE,
    CONSTRAINT "FK_SemesterSurveys_SurveyTemplates_SurveyTemplateId" FOREIGN KEY ("SurveyTemplateId")
        REFERENCES public."SurveyTemplates" ("SurveyTemplateId") ON DELETE RESTRICT
);
CREATE INDEX "IX_SemesterSurveys_SemesterId"       ON public."SemesterSurveys" ("SemesterId");
CREATE INDEX "IX_SemesterSurveys_SurveyTemplateId" ON public."SemesterSurveys" ("SurveyTemplateId");

CREATE TABLE public."CourseSectionSurveys" (
    "CourseSectionSurveyId" integer GENERATED BY DEFAULT AS IDENTITY,
    "SemesterSurveyId"      integer NOT NULL,
    "CourseSectionId"       integer NOT NULL,
    "LinkToken"             text NOT NULL DEFAULT gen_random_uuid()::text,  -- used to build the link + QR
    "StartTime"             timestamp with time zone NOT NULL,
    "EndTime"               timestamp with time zone NOT NULL,
    "CreatedAt"             timestamp with time zone NOT NULL,
    CONSTRAINT "PK_CourseSectionSurveys" PRIMARY KEY ("CourseSectionSurveyId"),
    CONSTRAINT "FK_CourseSectionSurveys_SemesterSurveys_SemesterSurveyId" FOREIGN KEY ("SemesterSurveyId")
        REFERENCES public."SemesterSurveys" ("SemesterSurveyId") ON DELETE CASCADE,
    CONSTRAINT "FK_CourseSectionSurveys_CourseSections_CourseSectionId" FOREIGN KEY ("CourseSectionId")
        REFERENCES public."CourseSections" ("CourseSectionId") ON DELETE CASCADE,
    CONSTRAINT "CK_CourseSectionSurveys_TimeRange" CHECK ("EndTime" > "StartTime")
);
CREATE UNIQUE INDEX "IX_CourseSectionSurveys_LinkToken" ON public."CourseSectionSurveys" ("LinkToken");
-- one section gets exactly one link per semester survey run
CREATE UNIQUE INDEX "IX_CourseSectionSurveys_SemesterSurveyId_CourseSectionId"
    ON public."CourseSectionSurveys" ("SemesterSurveyId", "CourseSectionId");
CREATE INDEX "IX_CourseSectionSurveys_CourseSectionId" ON public."CourseSectionSurveys" ("CourseSectionId");


-- ================= 10. STUDENT SUBMISSIONS (anonymous) =================

CREATE TABLE public."SurveyResponses" (
    "ResponseId"            integer GENERATED BY DEFAULT AS IDENTITY,
    "CourseSectionSurveyId" integer NOT NULL,
    "AdditionalComments"    text NULL,           -- optional free-text feedback
    "Score"                 numeric(4,2) NOT NULL,  -- average score, computed by backend on submit
    "SubmittedAt"           timestamp with time zone NOT NULL,
    CONSTRAINT "PK_SurveyResponses" PRIMARY KEY ("ResponseId"),
    CONSTRAINT "FK_SurveyResponses_CourseSectionSurveys_CourseSectionSurveyId" FOREIGN KEY ("CourseSectionSurveyId")
        REFERENCES public."CourseSectionSurveys" ("CourseSectionSurveyId") ON DELETE CASCADE
);
CREATE INDEX "IX_SurveyResponses_CourseSectionSurveyId" ON public."SurveyResponses" ("CourseSectionSurveyId");

CREATE TABLE public."SurveyResponseAnswers" (
    "ResponseId"    integer NOT NULL,
    "QuestionId"    integer NOT NULL,
    "SelectedValue" integer NOT NULL,        -- 1..5
    CONSTRAINT "PK_SurveyResponseAnswers" PRIMARY KEY ("ResponseId", "QuestionId"),
    CONSTRAINT "FK_SurveyResponseAnswers_SurveyResponses_ResponseId" FOREIGN KEY ("ResponseId")
        REFERENCES public."SurveyResponses" ("ResponseId") ON DELETE CASCADE,
    CONSTRAINT "FK_SurveyResponseAnswers_SurveyQuestions_QuestionId" FOREIGN KEY ("QuestionId")
        REFERENCES public."SurveyQuestions" ("QuestionId") ON DELETE RESTRICT,
    CONSTRAINT "CK_SurveyResponseAnswers_SelectedValue" CHECK ("SelectedValue" BETWEEN 1 AND 5)
);
CREATE INDEX "IX_SurveyResponseAnswers_QuestionId" ON public."SurveyResponseAnswers" ("QuestionId");
